cmake_minimum_required(VERSION 3.21)
project(led_projectm VERSION 4.1.6 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(ENABLE_PLAYLIST OFF CACHE BOOL "" FORCE)
set(ENABLE_SYSTEM_PROJECTM_EVAL OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
add_subdirectory("${PROJECTM_SOURCE}" engine)
add_executable(led-projectm renderer.cpp)
target_compile_definitions(led-projectm PRIVATE GL_SILENCE_DEPRECATION)
target_link_libraries(led-projectm PRIVATE libprojectM::projectM SDL2::SDL2 OpenGL::GL)
